cmake_minimum_required(VERSION 3.24)
project(geometry_wars_online LANGUAGES C)
set(SOLUNA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/soluna" CACHE PATH "Soluna source checkout")
if(NOT EXISTS "${SOLUNA_DIR}/extlua/extlua.c")
  message(FATAL_ERROR "Set SOLUNA_DIR to a Soluna source checkout")
endif()
if(EMSCRIPTEN)
  add_executable(websocket extlua/websocket.c extlua/radial_shape.c "${SOLUNA_DIR}/extlua/extlua.c" "${SOLUNA_DIR}/extlua/materialapi.c")
else()
  add_library(websocket MODULE extlua/websocket.c extlua/radial_shape.c "${SOLUNA_DIR}/extlua/extlua.c" "${SOLUNA_DIR}/extlua/materialapi.c")
endif()
target_include_directories(websocket PRIVATE "${SOLUNA_DIR}/3rd/lua" "${SOLUNA_DIR}/3rd" "${SOLUNA_DIR}/extlua" "${CMAKE_BINARY_DIR}")
target_compile_features(websocket PRIVATE c_std_11)
set_target_properties(websocket PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(EMSCRIPTEN)
  target_compile_options(websocket PRIVATE -pthread -fPIC -fwasm-exceptions)
  target_link_options(websocket PRIVATE -pthread -sSIDE_MODULE=1 -fwasm-exceptions)
  set_target_properties(websocket PROPERTIES SUFFIX ".wasm")
else()
  find_package(CURL 8.16 REQUIRED)
  target_link_libraries(websocket PRIVATE CURL::libcurl)
  if(APPLE)
    set_target_properties(websocket PROPERTIES SUFFIX ".so")
  endif()
endif()
if(MSVC)
  target_compile_options(websocket PRIVATE /W4)
else()
  target_compile_options(websocket PRIVATE -Wall -Wextra)
endif()
